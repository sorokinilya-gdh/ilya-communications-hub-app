function HubTogglePage20261004(previous,rows,checked){const next=new Set(previous);for(const row of rows)checked?next.add(row.id):next.delete(row.id);return next}
async function HubRunSelected20261004(rows,action,folder,onProgress){
 const succeeded=[],failed=[];let next=0;async function worker(){for(;;){const i=next++;if(i>=rows.length)return;const row=rows[i];try{
 if(action==='trash'&&folder==='trash'){const{data}=await Un.post('/folders/delete-batch',{folder:'trash',ids:[row.id]});if(!data.deleted?.includes(row.id))throw Error(data.failed?.map(x=>x.error).join('; ')||'Message unavailable or no longer in Trash');}
 else if(action==='important'||action==='not-important')await Un.post('/message/importance',{id:row.id,important:action==='important'});
 else if(action==='restore'&&folder==='spam')await Un.post('/message/not-spam',{id:row.id});
 else{if(action==='archive-important')await Un.post('/message/importance',{id:row.id,important:true});await Un.post('/message/action',{id:row.id,provider:row.provider.toLowerCase(),providerId:row.providerId||row.id.split(':').at(-1),mailbox:row.mailboxOwner,action:action==='archive-important'?'archive':action,sender:row.email,subject:row.subject});}
 succeeded.push(row.id);
 }catch(e){failed.push({id:row.id,error:String(e.message||e)})}onProgress?.(succeeded.length+failed.length,rows.length)}}
 await Promise.all(Array.from({length:Math.min(3,rows.length)},()=>worker()));return{succeeded,failed};
}
